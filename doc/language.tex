%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Abstract Syntax}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection{Values}
\[
\begin{array}{rcl}
v \;\;::=& n \\
        &|& b \\
        &|& C(v_1,\dots,v_n) \\
        &|& {f_1: v_1, \dots, f_m: v_m} \\
\end{array}
\]

where $n \in \mathbb{Z}$, $b \in \{\texttt{true}, \texttt{false}\}$, and $C$ is a constructor name.

\subsection{Types}

\[
\begin{array}{rcl}
T \;\;::=& \texttt{int} \\
        &|& \texttt{bool} \\
        &|& V \\ 
        && R
\end{array}
\]


Variant types are labeled sums:

\[
V ::= \langle C_1 : \overline{T_1}, \dots, C_n : \overline{T_n} \rangle
\]

where each $\overline{T_i}$ denotes a (possibly empty) sequence of types and $C_i$ are
constructor names.

Record types are labeled products:
\[
R ::= \{ f_1 : T_1, \dots, f_m : T_m \}
\]

where each $f_i$ is a field name and $T_i$ is its corresponding type.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Expressions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\[
\begin{array}{rcl}
e \;\;::=& v
\\ &|& e_1 \;\texttt{op}\; e_2
\\ &|& \texttt{!}\; e
\\ &|& x
\end{array}
\]

where $\texttt{op} \in \{ +, -, *, /, \land, \lor, ==, <, > \}$ and $x$ is a variable name.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Statements}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\[
\begin{array}{rcl}
s \;\;::=& \texttt{var } x : T
\\ &|& x := e
\\ &|& \texttt{if } e \;\{ s^* \}
\\ &|& \texttt{if } e \;\{ s^* \}\; \texttt{else } \{ s^* \}
\\ &|& \texttt{print } e
\\ &|& \texttt{variant } V = C_1(\overline{T_1}) \mid \dots \mid C_n(\overline{T_n})
\\ &|& \texttt{record } R = \{f_1 : T_1, \dots, f_m : T_m\} 
\end{array}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Typing Rules}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\[ 
\Gamma \vdash e : T
\]
% \[
% \Delta \vdash C(\overline{T}) : V
% \]
% \[
% \Delta \vdash \{ f_1 : T_1, \dots, f_m : T_m \} : R
% \]

\subsection{Base Types}

\begin{mathpar}
\inferrule*[right=T-Int]
{ }
{\Gamma \vdash n : \texttt{int}}

\inferrule*[right=T-Bool-True]
{ }
{\Gamma \vdash \texttt{true} : \texttt{bool}}

\inferrule*[right=T-Bool-False]
{ }
{\Gamma \vdash \texttt{false} : \texttt{bool}}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Variables}

\begin{mathpar}
\inferrule*[right=T-Var-Use]
{x : T \in \Gamma}
{\Gamma \vdash x : T}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Binary Operators}

\begin{mathpar}
\inferrule*[right=T-Arithmetic]
{\Gamma \vdash e_1 : \texttt{int} \\ \Gamma \vdash e_2 : \texttt{int}}
{\Gamma \vdash e_1 + e_2 : \texttt{int}}

\inferrule*[right=T-Logical]
{\Gamma \vdash e_1 : \texttt{bool} \\ \Gamma \vdash e_2 : \texttt{bool}}
{\Gamma \vdash e_1 \land e_2 : \texttt{bool}}

\inferrule*[right=T-Comparison]
{\Gamma \vdash e_1 : \texttt{int} \\ \Gamma \vdash e_2 : \texttt{int}}
{\Gamma \vdash e_1 \textless e_2 : \texttt{bool}}

\inferrule*[right=T-Equality]
{\Gamma \vdash e_1 : T \\ \Gamma \vdash e_2 : T}
{\Gamma \vdash e_1 == e_2 : \texttt{bool}}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Unary Operators}

\begin{mathpar}
\inferrule*[right=T-Not]
{\Gamma \vdash e : \texttt{bool}}
{\Gamma \vdash \texttt{not } e : \texttt{bool}}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{User-Defined Types}

\begin{mathpar}
\inferrule*[right=T-Variant-Constructor]
{
\Gamma \vdash e_1 : T_1 ,\cdots, e_n : T_n \\ 
C : (T_1,\dots,T_n) \in \langle C_1 : \overline{T_1}, \dots, C_n : \overline{T_n} \rangle = V
}
{
\Gamma \vdash C(e_1,\dots,e_n) : V
}
\end{mathpar}

% The constructor $C$ must be declared as part of variant $V$ with argument
% types $(T_1,\dots,T_n)$.

\begin{mathpar}
\inferrule*[right=T-Record-Value]
{
\Gamma \vdash e_1 : T_1 ,\cdots, e_m : T_m \\ 
R = \{ f_1 : T_1, \dots, f_m : T_m \}
}
{
\Gamma \vdash \{ f_1 : e_1, \dots, f_m : e_m \} : R
}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Statements}

% For each statement, typing rules indicate that the statement is well-typed
% under the given typing context.

% \begin{mathpar}
%     \inferrule*[right=T-Var-Decl]
%     { x \notin \text{dom}(\Gamma) }
%     {\Gamma \vdash \texttt{var } x : T \;\texttt{ok}}
% \end{mathpar}

% \begin{mathpar}
%     \inferrule*[right=T-Variant-Decl]
%     { V \notin \text{cod}(\Delta) }
%     {\Delta \vdash \texttt{variant } V = C_1(\overline{T_1}) \mid \dots \mid C_n(\overline{T_n}) \;\texttt{ok}}
% \end{mathpar}

\begin{mathpar}
    \inferrule*[right=T-Assign]
    {\Gamma \vdash e : T \\ x : T \in \Gamma
    }
    {\Gamma \vdash x := e \;\texttt{ok}}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=T-If]
    {\Gamma \vdash e : \texttt{bool} \\ \Gamma \vdash s^* \;\texttt{ok}}
    {\Gamma \vdash \texttt{if } e \{ s^* \} \;\texttt{ok}}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=T-If-Else]
    {\Gamma \vdash e : \texttt{bool} \\ \Gamma \vdash s_1^* \;\texttt{ok} \\ \Gamma \vdash s_2^* \;\texttt{ok}}
    {\Gamma \vdash \texttt{if } e \{ s_1^* \} \texttt{ else } \{ s_2^* \} \;\texttt{ok}}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=T-Print]
    {\Gamma \vdash e : T}
    {\Gamma \vdash \texttt{print } e \;\texttt{ok}}
\end{mathpar}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Operational Semantics}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The operational semantics are given as a big-step evaluation relation:
\[
\langle e, \sigma \rangle \to v
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Base Values}

\begin{mathpar}
\inferrule*[right=E-Int]
{ }
{\langle n, \sigma \rangle \to n}

\inferrule*[right=E-Bool]
{ }
{\langle \texttt{b}, \sigma \rangle \to \texttt{b}}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Variables}

\begin{mathpar}
\inferrule*[right=E-Var]
{\sigma(x) = v}
{\langle x, \sigma \rangle \to v}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Operations}

\begin{mathpar}
\inferrule*[right=E-Arithmetic]
{
\langle e_1, \sigma \rangle \to n_1 \\
\langle e_2, \sigma \rangle \to n_2
}
{
\langle e_1 + e_2, \sigma \rangle \to n_1 + n_2
}
\end{mathpar}

\begin{mathpar}
\inferrule*[right=E-Logical]
{
\langle e_1, \sigma \rangle \to b_1 \\
\langle e_2, \sigma \rangle \to b_2
}
{
\langle e_1 \land e_2, \sigma \rangle \to b_1 \land b_2
}
\end{mathpar}

\begin{mathpar}
\inferrule*[right=E-Not]
{
\langle e, \sigma \rangle \to b
}
{
\langle \texttt{!} e, \sigma \rangle \to \neg b
}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=E-Comparison]
{
\langle e_1, \sigma \rangle \to n_1 \\
\langle e_2, \sigma \rangle \to n_2
}
{
\langle e_1 \textless e_2, \sigma \rangle \to (n_1 < n_2)
}
\end{mathpar}

\begin{mathpar}
\inferrule*[right=E-Equality]
{
\langle e_1, \sigma \rangle \to v_1 \\
\langle e_2, \sigma \rangle \to v_2
}
{
\langle e_1 == e_2, \sigma \rangle \to (v_1 == v_2)
}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Variants}

\begin{mathpar}
\inferrule*[right=E-Variant-Constructor]
{
\langle e_1, \sigma \rangle \to v_1 \\
\cdots \\
\langle e_n, \sigma \rangle \to v_n
}
{
\langle C(e_1,\dots,e_n), \sigma \rangle \to C(v_1,\dots,v_n)
}
\end{mathpar}

\subsection{Records}

\begin{mathpar}
\inferrule*[right=E-Record-Value]
{
    \langle e_1 \sigma \rangle \to v_1 \\
    \cdots \\
    \langle e_n, \sigma \rangle \to v_n
}
{
    \langle \{ f_1 : e_1, \dots, f_n : e_n \}, \sigma \rangle \to \{ f_1 : v_1, \dots, f_n : v_n \}
}
\end{mathpar}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Statements}

Statement execution is defined as:
\[
\langle s, \sigma \rangle \to \sigma'
\]
\begin{mathpar}
\inferrule*[right=E-Assign]
{\langle e, \sigma \rangle \to v}
{\langle x := e, \sigma \rangle \to \sigma[x \mapsto v]}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=E-If-True]
    {\langle e, \sigma \rangle \to \texttt{true}\\ {\langle s^*, \sigma \rangle \to \sigma'}}
{\langle \texttt{if } e \{ s^* \}, \sigma \rangle \to \sigma'}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=E-If-False]
{\langle e, \sigma \rangle \to \texttt{false}}
{\langle \texttt{if } e \{ s^* \}, \sigma \rangle \to \sigma}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=E-If-Else-True]
    {\langle e, \sigma \rangle \to \texttt{true}\\ {\langle s_1^*, \sigma \rangle \to \sigma'}}
{\langle \texttt{if } e \{ s_1^* \} \texttt{ else } \{ s_2^* \}, \sigma \rangle \to \sigma'}
\end{mathpar}

\begin{mathpar}
    \inferrule*[right=E-If-Else-False]
    {\langle e, \sigma \rangle \to \texttt{false}\\ {\langle s_2^*, \sigma \rangle \to \sigma'}}
{\langle \texttt{if } e \{ s_1^* \} \texttt{ else } \{ s_2^* \}, \sigma \rangle \to \sigma'}
\end{mathpar}
